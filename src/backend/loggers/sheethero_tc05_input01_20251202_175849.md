all_files = list_all_workbooks()

# Step 2: Read data from both files using inspector_multi
file1_data = inspector_multi(all_files[0], "A1:K40", "Data")  # tc05_input01.xlsx 'Data' sheet

# Step 3: Convert to DataFrames
df1 = pd.DataFrame(file1_data[1:], columns=file1_data[0])  # Market Share Data
df2 = pd.DataFrame(file2_data[1:], columns=file2_data[0])  # Shipment Data

# Step 4: Clean and convert data types
df1['Market Share (%)'] = df1[['Vivo', 'Samsung', 'Xiaomi', 'Oppo', 'Apple', 'Realme', 'Lenovo', 'Others']].apply(pd.to_numeric, errors='coerce')  # Convert market share to numeric
df2['Total Shipments (Million)'] = pd.to_numeric(df2['C'], errors='coerce')  # Total shipments

# Step 5: Align the quarters for estimation calculations
# Define overlapping quarters
overlapping_times = df1[df1['Market Share (%)'].notna()]['Date'].unique()

# Initialize a DataFrame to store the results
result = pd.DataFrame(columns=['Time', 'Vivo (Unit shipment)', 'Samsung (Unit shipment)', 'Xiaomi (Unit shipment)', 'Oppo (Unit shipment)', 
                               'Apple (Unit shipment)', 'Realme (Unit shipment)', 'Lenovo (Unit shipment)', 'Others (Unit shipment)'])

# Fill in the result DataFrame with calculated estimates
for time in overlapping_times:
    if time in df2['Date'].values:
        shipments = df2.loc[df2['Date'] == time, 'Total Shipments (Million)'].values[0]  # Find total shipments for the corresponding quarter
        market_share_row = df1[df1['Date'] == time]
        
        if not market_share_row.empty:
            # Calculate unit shipments for each brand
            vivo_units = market_share_row['Vivo'].values[0] * shipments / 100
            samsung_units = market_share_row['Samsung'].values[0] * shipments / 100
            xiaomi_units = market_share_row['Xiaomi'].values[0] * shipments / 100
            oppo_units = market_share_row['Oppo'].values[0] * shipments / 100
            apple_units = market_share_row['Apple'].values[0] * shipments / 100
            realme_units = market_share_row['Realme'].values[0] * shipments / 100
            lenovo_units = market_share_row['Lenovo'].values[0] * shipments / 100
            others_units = market_share_row['Others'].values[0] * shipments / 100
            
            # Append results to DataFrame
            result = result.append({'Time': time, 
                                    'Vivo (Unit shipment)': vivo_units,
                                    'Samsung (Unit shipment)': samsung_units, 
                                    'Xiaomi (Unit shipment)': xiaomi_units,
                                    'Oppo (Unit shipment)': oppo_units,
                                    'Apple (Unit shipment)': apple_units,
                                    'Realme (Unit shipment)': realme_units,
                                    'Lenovo (Unit shipment)': lenovo_units,
                                    'Others (Unit shipment)': others_units}, 
                                   ignore_index=True)

# Step 6: Set up for output
# Convert result DataFrame to a list for output
result_data = [result.columns.tolist()] + result.values.tolist()
