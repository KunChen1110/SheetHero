from datetime import datetime, timedelta

# Step 1: Initialize the schedule with a starting time of 8:00 AM
start_time = datetime.strptime("8:00 AM", "%I:%M %p")
schedule = []

# Step 2: Define a function to schedule tasks
def schedule_tasks(df_tasks, df_deps):
    task_completion_times = {}
    
    # Sort tasks based on dependencies
    sorted_tasks = df_tasks['Task ID'].tolist()

    for task in sorted_tasks:
        # Check dependencies
        depends_on = df_deps[df_deps['Task ID_x000d_'] == task]['Depends on'].tolist()
        if depends_on:
            # Get the completion time of all dependencies
            latest_dependency_time = max(task_completion_times[dep] for dep in depends_on if dep in task_completion_times)
        else:
            # No dependencies, start at the initial start time
            latest_dependency_time = start_time
        
        # Calculate start time (max of latest dependency time and previous tasks start)
        task_start_time = latest_dependency_time
        task_duration = df_tasks.loc[df_tasks['Task ID'] == task, 'Duration (hours)'].values[0]

        # Calculate end time
        task_end_time = task_start_time + timedelta(hours=task_duration)

        # Store times and format them
        task_info = {
            'Task ID': task,
            'Task Name': df_tasks.loc[df_tasks['Task ID'] == task, 'Task Name'].values[0],
            'Priority': df_tasks.loc[df_tasks['Task ID'] == task, 'Priority'].values[0],
            'Start Time': task_start_time.strftime("%I:%M %p"),
            'End Time': task_end_time.strftime("%I:%M %p")
        }
        schedule.append(task_info)
        
        # Record the completion time
        task_completion_times[task] = task_end_time

# Step 3: Schedule the tasks
schedule_tasks(df_tasks, df_deps)

# Step 4: Convert the schedule to DataFrame for easier handling and output
df_schedule = pd.DataFrame(schedule)

# Step 5: Calculate total duration for all tasks
total_duration_hours = df_tasks['Duration (hours)'].sum()

# Show schedule and total duration
print(df_schedule)
print(f"Total duration of finishing all tasks (hours): {total_duration_hours}")
