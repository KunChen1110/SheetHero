dependencies_df['Task ID_x000d_'] = dependencies_df['Task ID_x000d_'].str.replace('_x000d_', '')

# Step 3: Create a dictionary to store the scheduled tasks
schedule = []
task_duration_dict = {row['Task ID']: row['Duration (hours)'] for _, row in tasks_df.iterrows()}

# Step 4: Track completed tasks
completed_tasks = set()

# Step 5: Initialize starting point
from datetime import datetime, timedelta

start_time = datetime.strptime("08:00", "%H:%M")

# Step 6: Schedule tasks based on dependencies
while True:
    current_tasks = dependencies_df[~dependencies_df['Task ID_x000d_'].isin(completed_tasks)]
    if current_tasks.empty:
        break  # Exit loop if there are no tasks left to schedule
    
    for index, row in current_tasks.iterrows():
        task_id = row['Task ID_x000d_']
        dependency = row['Depends on']
        
        # Check if dependency is satisfied
        if (pd.isna(dependency) or dependency in completed_tasks):
            # Calculate start and end times based on completed tasks
            start_time_for_task = start_time if task_id not in [r['Task ID'] for r in schedule] else start_time + timedelta(hours=task_duration_dict[task_id])
            end_time_for_task = start_time_for_task + timedelta(hours=task_duration_dict[task_id])
            schedule.append({
                'Task ID': task_id,
                'Task Name': tasks_df.loc[tasks_df['Task ID'] == task_id, 'Task Name'].values[0],
                'Priority': tasks_df.loc[tasks_df['Task ID'] == task_id, 'Priority'].values[0],
                'Start Time': start_time_for_task.strftime("%H:%M"),
                'End Time': end_time_for_task.strftime("%H:%M")
            })
            completed_tasks.add(task_id)
            start_time = end_time_for_task  # Update the start time for the next tasks
            
# Convert the schedule to DataFrame for better manipulation
schedule_df = pd.DataFrame(schedule)
