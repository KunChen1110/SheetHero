    combined_data[col] = pd.to_numeric(combined_data[col], errors='coerce')

# Determine number of skills by splitting the skills column
combined_data['number_of_skills'] = combined_data['Key Skills'].str.split(',').str.len()

# Step 5: Calculate the score according to the specified criteria
combined_data['Score'] = (
    0.5 * combined_data['YearsOfExperience'].fillna(0) +
    0.3 * combined_data['number_of_skills'].fillna(0) +
    0.2 * combined_data['Personality Score'].fillna(0)
)

# Step 6: Prepare the results and select relevant columns for output
