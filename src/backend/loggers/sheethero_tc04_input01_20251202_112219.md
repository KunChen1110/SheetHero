from collections import defaultdict, deque
import datetime

# Step 1: Build the graph from dependencies
graph = defaultdict(list)
indegree = {task: 0 for task in cleaned_tasks_df["Task ID"]}

# Populate the graph and indegree
for _, row in final_dependencies_df.iterrows():
    task = row["Task ID"]
    dependency = row["Depends on"] if pd.notna(row["Depends on"]) else None
    if dependency:
        graph[dependency].append(task)
        indegree[task] += 1

# Step 2: Find all root tasks (tasks with no dependencies)
queue = deque()
for task, degree in indegree.items():
    if degree == 0:
        queue.append(task)

# Step 3: Topological Sort using Kahn's Algorithm
topological_order = []
while queue:
    current_task = queue.popleft()
    topological_order.append(current_task)
    
    for neighbor in graph[current_task]:
        indegree[neighbor] -= 1
        if indegree[neighbor] == 0:
            queue.append(neighbor)

# Step 4: Verify that all tasks are included in the topological sort
if len(topological_order) != len(cleaned_tasks_df):
    raise Exception("Cycle detected or not all tasks are included in the ordering.")

# Step 5: Schedule the tasks based on the topological order
start_time = datetime.datetime.strptime("08:00", "%H:%M")
schedule = []

for task in topological_order:
    task_info = cleaned_tasks_df.loc[cleaned_tasks_df["Task ID"] == task].iloc[0]
    duration = task_info["Duration (hours)"]
    end_time = start_time + datetime.timedelta(hours=duration)
    
    schedule.append({
        "Task ID": task,
        "Task Name": task_info["Task Name"],
        "Priority": task_info["Priority"],
        "Start Time": start_time.strftime("%H:%M"),
        "End Time": end_time.strftime("%H:%M")
    })
    
    # Update the start time for the next task
    start_time = end_time

# Convert schedule to DataFrame for easier manipulation and viewing
schedule_df = pd.DataFrame(schedule)
