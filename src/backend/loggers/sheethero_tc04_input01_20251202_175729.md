# Step 3: Function for topological sorting
def topological_sort(tasks, dependencies):
    # Create an adjacency list and an in-degree count
    adjacency_list = {task: [] for task in tasks}
    in_degree = {task: 0 for task in tasks}

    # Build the graph
    for _, row in dependencies.iterrows():
        task_id = row['Task ID']
        depends_on = row['Depends on']
        if depends_on:  # If there is a dependency
            adjacency_list[depends_on].append(task_id)
            in_degree[task_id] += 1

    # Start with nodes that have no incoming edges
    queue = [task for task in tasks if in_degree[task] == 0]
    sorted_order = []

    while queue:
        task = queue.pop(0)  # Get the first task
        sorted_order.append(task)

        # Decrease the in-degree of neighbor tasks
        for neighbor in adjacency_list[task]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # Return the sorted order and check for cycle
    if len(sorted_order) != len(tasks):
        raise Exception("Graph has at least one cycle!")
    
    return sorted_order
