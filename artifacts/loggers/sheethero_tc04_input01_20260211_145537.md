from collections import defaultdict, deque

# Step 5: Build the graph for topological sorting
# Create a graph and a list to keep track of in-degrees
graph = defaultdict(list)
in_degree = {task_id: 0 for task_id in tasks_df['Task ID']}

# Populate the graph and in-degrees based on the dependencies DataFrame
for _, row in dependencies_df.iterrows():
    task_id = row['Task ID']
    depends_on = row['Depends on']
    
    if depends_on:  # If there is a dependency
        graph[depends_on].append(task_id)
        in_degree[task_id] += 1

# Step 6: Perform topological sort using Kahn's algorithm
topological_order = []
queue = deque()

# Start with tasks that have no dependencies (in_degree == 0)
for task_id in in_degree:
    if in_degree[task_id] == 0:
        queue.append(task_id)

while queue:
    current_task = queue.popleft()
    topological_order.append(current_task)
    
    for neighbor in graph[current_task]:
        in_degree[neighbor] -= 1
        if in_degree[neighbor] == 0:
            queue.append(neighbor)

# Step 7: Create the schedule based on topological order
start_time = pd.Timestamp('08:00')
total_duration = 0
schedule = []

for task_id in topological_order:
    task_row = tasks_df[tasks_df['Task ID'] == task_id].iloc[0]
    duration = float(task_row['Duration (hours)'])
    
    start_time_str = start_time.strftime('%H:%M')
    end_time = start_time + pd.Timedelta(hours=duration)
    
    # Add task details to the schedule
    schedule.append({
        'Task ID': task_id,
        'Task Name': task_row['Task Name'],
        'Priority': task_row['Priority'],
        'Start Time': start_time_str,
        'End Time': end_time.strftime('%H:%M')
    })
    
    # Move start time to the end of the current task
    start_time = end_time
    total_duration += duration

# Convert schedule to DataFrame for easier visualization
schedule_df = pd.DataFrame(schedule)

# Display the final schedule and total duration
total_duration, schedule_df
